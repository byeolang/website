62 struct subLess {
63 nbool operator()(const std::string& l, const std::string& r) const {
64 const nbool lIdx = isIdx(l), rIdx = isIdx(r);
65 if(lIdx != rIdx) return lIdx; // indices sort ahead of names.
66 if(!lIdx) return l < r; // names keep their lexicographic order.
67 // indices carry no leading zero, so length-then-lexicographic == numeric.
68 return l.size() != r.size() ? l.size() < r.size() : l < r;
69 }
70
71 static nbool isIdx(const std::string& it) {
72 if(it.empty()) return false;
73 for(nchar c: it)
74 if(c < '0' || c > '9') return false;
75 return true;
76 }
77 };
